#include <iostream> 
#include <optional> 
#include <vector> 
// 查找vector中第一个偶数，找到返回值，否则返回空 
std::optional<int> find_even(const std::vector<int>& vec) { 
	for (int num : vec) { 
		if (num % 2 == 0) { 
			return num; // 直接返回值，编译器自动封装为optional
		} 
	} return std::nullopt; // 无值，返回空 
} 
int main() { 
	std::vector<int> v1 = {1,3,5,6,9}; 
	std::vector<int> v2 = {1,3,5,7}; 
	auto res1 = find_even(v1); 
	if (res1) { // 直接判断是否有值 
		std::cout << "v1找到偶数：" << *res1 << std::endl; 
	} 
	auto res2 = find_even(v2); // 无值时返回默认值-1，避免判空，推荐用法 
	std::cout << "v2找到的偶数：" << res2.value_or(-1) << std::endl; return 0; 
}

### 结构化绑定（structured bindings，auto \[..]，C++17）

```cpp
#include <iostream>
#include <tuple>
#include <map>
#include <array>

// 简单自定义结构体（聚合类型，无自定义构造函数）
struct Student {
    std::string name;
    int age;
    double score;
};

int main() {
    // 1. 拆包std::pair
    std::pair<std::string, int> p = {"张三", 20};
    auto [name1, age1] = p;
    std::cout << "pair拆包：" << name1 << "，" << age1 << std::endl;

    // 2. 拆包std::tuple
    std::tuple<std::string, int, double> t = {"李四", 19, 95.5};
    auto [name2, age2, score2] = t;
    std::cout << "tuple拆包：" << name2 << "，" << age2 << "，" << score2 << std::endl;

    // 3. 遍历map（核心常用场景，替代it->first/it->second）
    std::map<std::string, int> fruit_price = {{"苹果",5}, {"香蕉",3}, {"橙子",4}};
    for (const auto& [fruit, price] : fruit_price) {
        std::cout << fruit << "：" << price << "元" << std::endl;
    }

    // 4. 拆包数组+引用绑定（可修改原对象）
    std::array<int, 3> arr = {1,2,3};
    auto& [a, b, c] = arr;
    c = 10; // 修改绑定变量，原数组也会被修改
    std::cout << "数组修改后：" << arr[2] << std::endl;

    // 5. 拆包自定义聚合结构体
    Student s = {"王五", 18, 88.0};
    auto [name3, age3, score3] = s;
    std::cout << "结构体拆包：" << name3 << "，" << age3 << "，" << score3 << std::endl;
    return 0;
}
```

#include <iostream>
#include <tuple>
#include <string>

// 核心场景：函数返回多个不同类型的值（姓名+年龄+是否及格）
std::tuple<std::string, int, bool> get_student_info() {
    // 直接返回，编译器自动封装；也可用std::make_tuple("赵六", 20, true)
    return {"赵六", 20, true};
}

int main() {
    // 方式1：C++11传统拆包（std::get<N>），兼容性好但繁琐
    auto t1 = get_student_info();
    std::string name1 = std::get<0>(t1);
    int age1 = std::get<1>(t1);
    bool pass1 = std::get<2>(t1);
    std::cout << "传统拆包：" << name1 << "，" << age1 << "，" << (pass1 ? "及格" : "不及格") << std::endl;

    // 方式2：C++17结构化绑定（推荐，简洁）
    auto [name2, age2, pass2] = get_student_info();
    std::cout << "结构化绑定拆包：" << name2 << "，" << age2 << "，" << (pass2 ? "及格" : "不及格") << std::endl;

    // 修改变量值
    auto t2 = std::make_tuple(1, 2.5, "test");
    std::get<2>(t2) = "modify";
    std::cout << "修改后tuple第3个值：" << std::get<2>(t2) << std::endl;
    return 0;
}

### emplace_back /initializer-list 用法（C++11）

```cpp
#include <iostream>
#include <vector>
#include <string>

// 自定义类（添加构造函数打印，直观看到对象创建/拷贝）
class Person {
public:
    std::string name;
    int age;
    // 构造函数
    Person(std::string n, int a) : name(n), age(a) {
        std::cout << "Person构造：" << name << std::endl;
    }
    // 拷贝构造函数
    Person(const Person& p) : name(p.name), age(p.age) {
        std::cout << "Person拷贝：" << name << std::endl;
    }
    // 移动构造函数（C++11，可省略，编译器自动生成）
    Person(Person&& p) noexcept : name(std::move(p.name)), age(p.age) {
        std::cout << "Person移动：" << name << std::endl;
    }
};

int main() {
    // 场景1：initializer-list 列表初始化容器（核心用法）
    std::vector<int> vec1 = {1,2,3,4,5}; // 传统列表初始化
    std::vector<std::string> vec2{"apple", "banana", "orange"}; // 直接花括号，推荐
    std::cout << "vec2大小：" << vec2.size() << std::endl;

    // 场景2：push_back vs emplace_back 对比（核心高效点）
    std::vector<Person> vec_p;
    vec_p.reserve(2); // 预留空间，避免容器扩容影响测试

    std::cout << "===== push_back 插入 =====" << std::endl;
    // push_back：先构造临时对象，再移动/拷贝到容器
    vec_p.push_back(Person("张三", 20));

    std::cout << "===== emplace_back 插入 =====" << std::endl;
    // emplace_back：直接传递构造参数，就地构造，无临时对象
    vec_p.emplace_back("李四", 21);

    return 0;
}
```
