            info = new UserInfo;

            info->m_sockfd = clientfd;
            strcpy( info->m_userName, strName.c_str() );

            strcpy( rs.name, strName.c_str() );
            rs.userid = id;
            //id 捆绑套结字
            m_mapFdToUserInfo.insert( id, info );
            //返回结果id name result
